[wsl2]
# 1. メモリの上限設定（パソコンの総メモリの 33% 〜 50% が目安）
# 例：総メモリ16GBなら 8GB、32GBなら 16GB
memory=16GB

# 2. メモリの自動解放（WSL内でメモリを使わなくなったらWindowsに戻す設定）
autoMemoryReclaim=dropcache

# 3. ネットワークモード（WindowsとIPアドレスを共有し、localhostでの接続トラブルを激減させる）
networkingMode=mirrored

# 4. Windows側のホスト名を自動でWSLの /etc/hosts に反映させる
localhostForwarding=true

# 5. コアサイズ（クラッシュ時のダンプファイル生成を抑止しディスクを節約）
coreDump=false

[boot]
systemd=true
